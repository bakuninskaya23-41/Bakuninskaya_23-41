const TOTAL = 226;

// Найдём текущую заметку и таблицу в ней
let raw = dv.current().file.content;

// Ищем все строки таблицы, которые содержат значения
let lines = raw.split("\n").filter(l => l.trim().startsWith("|") && !l.includes("---"));

// Пропускаем заголовки
lines.shift(); // заголовок
lines.shift(); // разделитель

// Считаем
let counts = { "+": 0, "-": 0, "+-": 0, "empty": 0, "other": 0 };

for (let line of lines) {
  let cols = line.split("|").map(s => s.trim());
  let status = cols[5] || "";  // 5 — это 6-й столбец (Статус)

  if (status === "+") counts["+"]++;
  else if (status === "-") counts["-"]++;
  else if (status === "+-") counts["+-"]++;
  else if (status === "") counts["empty"]++;
  else counts["other"]++;
}

function bar(count) {
  let pct = count / TOTAL;
  let filled = Math.round(pct * 30);
  let empty = 30 - filled;
  return "▉".repeat(filled) + "—".repeat(empty) + ` ${Math.round(pct*100)}%`;
}

dv.header(3, `Статистика заполнения статусов (из ${TOTAL})`);
dv.paragraph(`
**+**  ${counts["+"]} / ${TOTAL} → ${bar(counts["+"])}  
**-**  ${counts["-"]} / ${TOTAL} → ${bar(counts["-"])}  
**+-** ${counts["+-"]}/ ${TOTAL} → ${bar(counts["+-"])}  
**пусто** ${counts["empty"]}/ ${TOTAL} → ${bar(counts["empty"])}  
**другое** ${counts["other"]}/ ${TOTAL} → ${bar(counts["other"])}
`);
