(async () => {
  const TOTAL = 226;  // ваше общее число строк

  // 1. Получаем путь текущего файла
  const path = dv.current().file.path;

  // 2. Находим файл в хранилище и читаем его содержимое
  const file = this.app.vault.getAbstractFileByPath(path);
  const raw = await this.app.vault.read(file);

  // 3. Разбиваем по строкам и фильтруем таблицу
  let lines = raw.split("\n")
    .filter(l => l.trim().startsWith("|") && !l.includes("---"));

  // 4. Убираем первую строку (заголовок) и вторую (разделитель)
  if (lines.length >= 1) {
    lines = lines.slice(1);
  } else {
    dv.paragraph("❗ Не найдена таблица с `| … | … |`");
    return;
  }

  // 5. Считаем статусы
  let counts = { "+": 0, "-": 0, "+-": 0, "empty": 0, "other": 0, "?":0 };
  for (let line of lines) {
    // разбиваем по | и убираем пробелы
    let cols = line.split("|").map(s => s.trim());
    // 5-й индекс (cols[5]) — это шестой столбец, где у вас Status
    let status = cols[5] || "";
    if (status === "+") counts["+"]++;
    else if (status === "-") counts["-"]++;
    else if (status === "+-") counts["+-"]++;
    else if (status === "") counts["empty"]++;
     else if (status === "?") counts["?"]++;
    else counts["other"]++;
  }

  // 6. Функция для прогресс-бара
  function bar(count) {
    let pct = count / TOTAL;
    let filled = Math.round(pct * 30);
    let empty = 30 - filled;
    return "▉".repeat(filled) + "—".repeat(empty) + ` ${Math.round(pct*100)}%`;
  }

  // 7. Выводим
  dv.header(3, `Статистика сбора (из ${TOTAL})`);
  dv.paragraph(`
**+**  ${counts["+"]} / ${TOTAL} → ${bar(counts["+"])}  
**-**  ${counts["-"]} / ${TOTAL} → ${bar(counts["-"])}  
**+-** ${counts["+-"]}/ ${TOTAL} → ${bar(counts["+-"])}  
**пусто** ${counts["empty"]}/ ${TOTAL} → ${bar(counts["empty"])}  
**другое** ${counts["other"]}/ ${TOTAL} → ${bar(counts["other"])}
**?** ${counts["?"]}/ ${TOTAL} → ${bar(counts["?"])}
`);
})();
