```dataviewjs
// 1. Общая константа
const TOTAL = 226;  // поменяйте, если число записей меняется

// 2. Собираем все записи с ключом `Статус`
let pages = dv.pages('"."')  // или указать конкретную папку
  .where(p => p.Статус);

// 3. Считаем статусы
let counts = { "+":0, "-":0, "+-":0, "other":0 };
for (let p of pages) {
  let s = p.Статус.toString().trim();
  if (counts[s] !== undefined) counts[s]++; 
  else counts.other++;
}

// 4. Функция для прогресс-бара (макс. длина 30 символов)
function bar(count) {
  let pct = count / TOTAL;
  let filled = Math.round(pct * 30);
  let empty  = 30 - filled;
  return "▉".repeat(filled) + "—".repeat(empty) + ` ${Math.round(pct*100)}%`;
}

// 5. Выводим результаты
dv.header(3, `Статистика заполнения статусов (из ${TOTAL})`);
dv.paragraph(`
**+**  ${counts["+"]} / ${TOTAL} → ${bar(counts["+"])}  
**-**  ${counts["-"]} / ${TOTAL} → ${bar(counts["-"])}  
**+-** ${counts["+-"]}/ ${TOTAL} → ${bar(counts["+-"])}  
**другие** ${counts.other}/ ${TOTAL} → ${bar(counts.other)}
`);
```
